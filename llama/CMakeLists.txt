# Builds the patched llama.cpp and the Cale shim into one shared library: cale_llama.
# usage: cmake -S llama -B build -DLLAMA_CPP_DIR=/path/to/patched/llama.cpp -DCMAKE_BUILD_TYPE=Release
cmake_minimum_required(VERSION 3.18)
project(cale_llama CXX C)
set(LLAMA_CPP_DIR "" CACHE PATH "the llama.cpp source folder, with llama/patches applied")
if(NOT LLAMA_CPP_DIR)
  message(FATAL_ERROR "set -DLLAMA_CPP_DIR=<llama.cpp folder>")
endif()
# CALE_BACKENDS_DL=ON: ggml's compute backends are separate libraries loaded at run time (CPU variants for each
# instruction set, plus GPU backends such as Vulkan found next to cale_llama). One install then runs on any computer.
# OFF: everything is linked into one library, CPU only (the Phase 5 build).
option(CALE_BACKENDS_DL "load compute backends (CPU variants, GPU) at run time" OFF)
if(CALE_BACKENDS_DL)
  set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
  set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
  set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
else()
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
# One binary that runs on any x86-64 computer with AVX2 (2013 and later). Not tuned to the build machine.
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
  set(GGML_AVX2 ON CACHE BOOL "" FORCE)
  set(GGML_FMA ON CACHE BOOL "" FORCE)
  set(GGML_F16C ON CACHE BOOL "" FORCE)
  set(GGML_BMI2 ON CACHE BOOL "" FORCE)
endif()
if(WIN32 AND CALE_BACKENDS_DL)
  set(CMAKE_SHARED_LIBRARY_PREFIX "")    # file names ggml-base.dll, llama.dll: the official GPU backend DLLs load them by these names
endif()
add_subdirectory(${LLAMA_CPP_DIR} llama.cpp)
add_library(cale_llama SHARED shim/cale_llama.cpp)
target_include_directories(cale_llama PRIVATE ${LLAMA_CPP_DIR}/include ${LLAMA_CPP_DIR}/ggml/include)
target_link_libraries(cale_llama PRIVATE llama)
if(CALE_BACKENDS_DL)
  target_compile_definitions(cale_llama PRIVATE CALE_BACKENDS_DL)
  target_include_directories(cale_llama PRIVATE ${LLAMA_CPP_DIR}/common)
  target_link_libraries(cale_llama PRIVATE llama-common)
endif()
if(WIN32)
  # Windows 8 or newer (PrefetchVirtualMemory). One DLL that needs no MinGW runtime DLLs next to it. File name: cale_llama.dll
  target_compile_definitions(cale_llama PRIVATE _WIN32_WINNT=0x0A00)
  target_link_libraries(cale_llama PRIVATE psapi)
  if(CALE_BACKENDS_DL)
    target_link_options(cale_llama PRIVATE -static-libgcc -static-libstdc++)
  else()
    target_link_options(cale_llama PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  set_target_properties(cale_llama PROPERTIES PREFIX "")
endif()
option(CALE_STOCK "build against unpatched llama.cpp (for the full-size equality check)" OFF)
if(CALE_STOCK)
  target_compile_definitions(cale_llama PRIVATE CALE_STOCK)
endif()
set_target_properties(cale_llama PROPERTIES CXX_VISIBILITY_PRESET hidden)
